PokeAPI — Selección por clima

El Pokémon que aparece depende de temperatura, viento y si es de día o noche

Lógica de pesos

  • Temp <0°C → Ice +5
  • Temp 20-30°C → Fire +3, Electric +2
  • Viento >25 m/s → Flying +5
  • Code 0 sol → Fire +3, Flying +2
  • Noche → Ghost +4, Dark +3

Endpoint consumido

GET pokeapi.co/api/v2/type/{tipo}

Selección aleatoria ponderada → petición al Pokémon → limpieza de datos

class PokeAPI:

  def calcular_pesos(self, temperature, windspeed,
                     weather_code, is_day):
    weather_code = weather_code / 100
    windspeed    = (windspeed / 1000) * 3600

    weights = {
        "ice": 0, "water": 0, "steel": 0,
        "grass": 0, "normal": 0, "ground": 0,
        "fire": 0, "electric": 0, "fighting": 0,
        "dragon": 0, "flying": 0, "ghost": 0,
        "psychic": 0, "dark": 0
    }

    # ── TEMPERATURA ──────────────────────────
    if temperature < 0:
        weights["ice"] += 5
    elif temperature < 10:
        weights["water"] += 4; weights["steel"] += 2
    elif temperature < 20:
        weights["grass"] += 3; weights["normal"] += 2
    elif temperature < 30:
        weights["fire"] += 3; weights["electric"] += 2
    else:
        weights["fire"] += 3; weights["dragon"] += 3

    # ── VIENTO ───────────────────────────────
    if windspeed > 25:
        weights["flying"] += 5; weights["dragon"] += 2
    elif windspeed > 10:
        weights["flying"] += 3; weights["electric"] += 2

    # ── WEATHER CODE ─────────────────────────
    if   weather_code == 0:         # sol despejado
        weights["fire"] += 3; weights["flying"] += 2
    elif weather_code in [1,3,8]:    # nubes / parcial
        weights["grass"] += 3; weights["ghost"] += 1
    elif weather_code == 2:         # lluvia
        weights["electric"] += 4; weights["water"] += 3
    elif weather_code == 7:         # niebla
        weights["ghost"] += 3; weights["psychic"] += 2
    elif weather_code == 6:         # nieve
        weights["ice"] += 5; weights["water"] += 1

    # ── DÍA / NOCHE ──────────────────────────
    if is_day == 1:
        weights["normal"] += 3; weights["grass"] += 2
        weights["ghost"]  -= 1; weights["dark"]  -= 1
    else:
        weights["ghost"] += 4; weights["dark"] += 3
        weights["grass"] -= 1; weights["fire"] -= 1

    return {t: max(0, w) for t, w in weights.items()}


  def seleccionar_tipo(self, weights):
    tipos = list(weights.keys())
    probs = list(weights.values())
    return random.choices(tipos, weights=probs, k=1)[0]


  @staticmethod
  def obtener_pokemon_por_tipo(tipo, numPokemons=1):
    response = requests.get(
        f"https://pokeapi.co/api/v2/type/{tipo}")
    lista = response.json()["pokemon"]
    if numPokemons == 1:
        entry = random.choice(lista)
        pokemon_response = requests.get(entry["pokemon"]["url"])
        return PokeAPI.limpiar_pokemon(pokemon_response.json())
    # Varios: sin repetición
    seleccionados = random.sample(lista, min(numPokemons, len(lista)))
    return [PokeAPI.limpiar_pokemon(
                requests.get(e["pokemon"]["url"]).json())
            for e in seleccionados]

Jonathan · OpenWeatherMap

OpenWeatherMap — Datos meteorológicos

Pronóstico horario con cálculo automático de si el sol es visible

¿Qué devuelve?

  • Temperatura en °C (units=metric)
  • Velocidad del viento
  • Código de clima (weather_code)
  • sun_visible → bool calculado
  • Sunrise / Sunset legibles

API key

Guardada en .env como
OPENWEATHER_API_KEY
cargada con os.getenv()

sun_visible

Compara datetime.now() con los timestamps Unix de sunrise/sunset que devuelve la API. Resultado: True/False. PokeAPI lo usa para potenciar tipos nocturnos.

external_servicies/openweathermap.py
from datetime import datetime
import requests

class OpenWeatherMap:
  def __init__(self, api_key):
    self.base_url = "https://api.openweathermap.org/data/2.5/forecast"
    self.api_key  = api_key

  def get_coord_forecast(self, coord: dict,
                          hours_to_forecast: int = 1) -> dict:

    querystring = {
        "lat":   coord.get('lat'),
        "lon":   coord.get('lon'),
        "appid": self.api_key,
        "cnt":   hours_to_forecast,
        "units": "metric"
    }

    response = requests.get(self.base_url, params=querystring)
    data     = response.json()

    if response.status_code != 200:
        return data   # devuelve el error de la API tal cual

    city = data.get('city')
    if city:
        city['sun_visible'] = False

        sunrise = city.get('sunrise')
        sunset  = city.get('sunset')

        if sunrise:
            city['sunrise_readable'] = datetime.fromtimestamp(sunrise)
        if sunset:
            city['sunset_readable']  = datetime.fromtimestamp(sunset)

        # ¿Está el sol visible ahora mismo?
        if ('sunrise_readable' in city and
                'sunset_readable' in city):
            city['sun_visible'] = (
                city['sunrise_readable']
                < datetime.now()
                < city['sunset_readable']
            )

        data['city'] = city

    return data

import requests
from django.conf import settings

class GooglePlacesServices:

  @staticmethod
  def search_places_nearBy(lat, long):
    """Busca restaurantes en un radio de 3 km"""
    url = "https://places.googleapis.com/v1/places:searchNearby"

    headers = {
        "Content-Type"   : "application/json",
        "X-Goog-Api-Key"  : settings.GOOGLE_PLACES_API_KEY,
        "X-Goog-FieldMask": "places.displayName,"
                            "places.location,"
                            "places.currentOpeningHours.openNow,"
                            "places.formattedAddress,"
                            "places.rating,"
                            "places.priceRange",
    }
    payload = {
        "includedTypes"   : ["restaurant"],
        "maxResultCount"  : 5,
        "locationRestriction": {
            "circle": {
                "center": {
                    "latitude" : lat,
                    "longitude": long,
                },
                "radius": 3000.0
            }
        }
    }
    response = requests.post(url, json=payload, headers=headers)

    if   response.status_code == 200: return response.json()
    elif response.status_code == 401: return {"error": "API key inválida"}
    elif response.status_code == 400: return {"error": "Petición incorrecta"}
    else: return {"error": f"Error: {response.status_code}"}


  @staticmethod
  def search_places_by_text(query, location_bias=None):
    """Búsqueda por texto libre"""
    url = "https://places.googleapis.com/v1/places:searchText"

    headers = {
        "Content-Type"   : "application/json",
        "X-Goog-Api-Key"  : settings.GOOGLE_PLACES_API_KEY,
        "X-Goog-FieldMask": "places.id,places.displayName,"
                            "places.formattedAddress,"
                            "places.currentOpeningHours.openNow,"
                            "places.rating",
    }
    payload = {
        "textQuery"     : query,
        "languageCode"  : "es",
        "maxResultCount": 5,
    }
    if location_bias:
        payload['locationBias'] = location_bias

    response = requests.post(url, json=payload, headers=headers)
    if response.status_code == 200: return response.json()
    else: return {"error": f"Error: {response.status_code}"}

Wikidata — Crimen histórico del día

Consulta SPARQL — el crimen más conocido que pasó en esta fecha

Tipos buscados

Q132821 crime Q318296 murder Q329425 homicide Q891854 assassination

Estrategia

  • Filtra por mismo día y mes (cualquier año)
  • Ordena por sitelinks DESC → el más famoso
  • LIMIT 1 → un crimen
  • Sin resultado → devuelve null
  • Rate-limit → lanza excepción
external_servicies/wikidata.py
import requests

class wikidata_crime:
  def __init__(self, user_agent):
    self.user_agent = user_agent

  def get_crime(self, actual_date):
    try:
        # Parsear "YYYY-MM-DD"
        if type(actual_date) == str:
            current_day   = str(actual_date[8:10])
            current_month = str(actual_date[5:7])
        else:
            current_day   = str(actual_date.day)
            current_month = str(actual_date.month)

        sparql_query = """
SELECT ?itemLabel ?itemDescription ?enlaces WHERE {
  VALUES ?crimen {
    wd:Q132821 wd:Q318296
    wd:Q329425 wd:Q891854
  }
  { ?item wdt:P31  ?crimen. }
  UNION
  { ?item wdt:P509 ?crimen. }

  ?item wdt:P585 | wdt:P570 ?fecha.
  FILTER(MONTH(?fecha) = [month] && DAY(?fecha) = [day])

  OPTIONAL { ?item wikibase:sitelinks ?enlaces. }
  SERVICE wikibase:label {
    bd:serviceParam wikibase:language "es,en".
  }
}
ORDER BY DESC(?enlaces)
LIMIT 1"""

        query_final = (sparql_query
            .replace("[month]", current_month)
            .replace("[day]",   current_day))

        response = requests.get(
            "https://query.wikidata.org/sparql",
            headers={"User-Agent": self.user_agent},
            params={"query": query_final, "format": "json"}
        )
        devuelve = response.json()
        crimen   = devuelve.get("results", {}).get("bindings", [])

        if not crimen:
            return {"itemLabel": None,
                    "itemDescription": None,
                    "enlaces": None}

        # Aplanar: { key: { "value": ... } } → { key: value }
        crimenLimpio = {}
        for key in crimen[0].keys():
            crimenLimpio[key] = crimen[0][key]["value"]
        return crimenLimpio

    except requests.exceptions.RequestException:
        raise Exception("Has alcanzado el RageLimit")

PokeCrimeWeatherView — el endpoint de producción

GET/api/pokecrimenweather/?lat=40.4&lon=-3.7&date=2026-03-22&ApiKey=…

Permiso requerido

hasAPIKey AllowAny

La API key va en la query: ?ApiKey=<token>

Flujo de datos

  1. Validar lat / lon
  2. Clima → OpenWeatherMap
  3. Pesos → tipo → Pokémon (caché 7d)
  4. Crimen del día → Wikidata (caché hasta medianoche)
  5. Restaurantes → Google Places → serializar
  6. Tweet → Gemini
  7. Devolver JsonResponse completo

Respuesta JSON

{
  "code":       200,
  "msg":        "<tweet Gemini>",
  "crime":      { "itemLabel": ... },
  "pokemon":    { "name": ..., "types": [...] },
  "clima":      { ... },
  "restaurant": [{ "name": ... }]
}
api/views.py — PokeCrimeWeatherView
class PokeCrimeWeatherView(APIView):
  permission_classes = [permissions.AllowAny, hasAPIKey]

  def get(self, request):
    date = request.query_params.get(
        'date', datetime.datetime.now().strftime("%Y-%m-%d"))
    lat  = request.query_params.get('lat')
    lon  = request.query_params.get('lon')

    if not lat or not lon:
        return HttpResponse(
            json.dumps({"error": "Faltan 'lat' y 'lon'"}),
            status=400)

    # ── 1. CLIMA ─────────────────────────────────────
    try:
        coord     = {"lat": lat, "lon": lon}
        data_clima = OpenWeatherMap(api_key).get_coord_forecast(coord)
    except Exception:
        return HttpResponse(
            json.dumps({"msg": "Error al obtener clima."}),
            status=400)

    # ── 2. POKÉMON (con caché 7 días) ───────────────
    try:
        poke      = PokeAPI()
        pesos     = poke.calcular_pesos(
                        data_clima["list"][0]["main"]["temp"],
                        data_clima["list"][0]["wind"]["speed"],
                        data_clima["list"][0]["weather"][0]["id"],
                        data_clima["city"]["sun_visible"])
        poke_type    = poke.seleccionar_tipo(pesos)
        pokemon_elegido = RedisCache.cache_get_poke_list(poke_type)
    except Exception as e:
        return HttpResponse(
            json.dumps({"msg": "Error Pokémon: " + str(e)}),
            status=400)

    # ── 3. VALIDAR FECHA ────────────────────────────
    try:
        date = datetime.datetime.strptime(date, "%Y-%m-%d").strftime("%Y-%m-%d")
    except (ValueError, TypeError):
        return HttpResponse(
            json.dumps({"msg": "Usa el formato YYYY-MM-DD."}),
            status=400)

    # ── 4. CRIMEN (con caché hasta medianoche) ──────
    try:
        crime = RedisCache.cache_get_info_crimen(date)
    except Exception:
        return HttpResponse(
            json.dumps({"msg": "Error Wikidata."}), status=400)

    # ── 5. RESTAURANTES + SERIALIZER ────────────────
    restaurante = None
    try:
        raw = GooglePlacesServices.search_places_nearBy(
                  lat=float(lat), long=float(lon))
        if raw and 'places' in raw:
            serializer  = PlaceSerializer(raw['places'], many=True)
            restaurante = serializer.data
    except Exception as e:
        return HttpResponse(
            json.dumps({"msg": "Error Places: " + str(e)}),
            status=500)

    # ── 6. GEMINI ────────────────────────────────────
    try:
        gemini_response = Gemini().generar_broma(
            pokemon_elegido,
            data_clima,
            crime,
            restaurante[0] if restaurante else None
        )
    except Exception as e:
        return HttpResponse(
            json.dumps({"msg": "Error Gemini: " + str(e)}),
            status=400)

    # ── 7. RESPUESTA ─────────────────────────────────
    return JsonResponse({
        "code":       status.HTTP_200_OK,
        "msg":        f'{gemini_response}',
        "crime":      crime,
        "pokemon":    pokemon_elegido,
        "clima":      data_clima,
        "restaurant": restaurante,
    }, status=status.HTTP_200_OK)

class RegisterView(generics.CreateAPIView):
  queryset         = User.objects.all()
  serializer_class = UserRegisterSerializer
  permission_classes = [permissions.AllowAny]

  def perform_create(self, serializer):
    password = self.request.data.get("password")
    user = serializer.save()
    if password:
        user.set_password(password)   # hashear
        user.save()
        UserAPIKey.objects.create(user=user)  # crear key

class LoginView(APIView):
  permission_classes = [permissions.AllowAny]
  serializer_class   = LoginSerializer

  def post(self, request):
    username = request.data.get("username")
    password = request.data.get("password")
    user = authenticate(request, username=username, password=password)
    if user is not None:
        login(request, user)
        return Response({"message": "Login successful"})
    return Response({"error": "Invalid credentials"}, status=400)

class PlaceSerializer(serializers.Serializer):
  name     = serializers.SerializerMethodField()
  address  = serializers.SerializerMethodField()
  rating   = serializers.FloatField(required=False)
  location = serializers.SerializerMethodField()
  openNow  = serializers.SerializerMethodField()

  def get_name(self, obj):
    return obj.get('displayName', {}).get('text', 'Sin nombre')

  def get_address(self, obj):
    return obj.get('formattedAddress', 'N/D')

  def get_location(self, obj):
    loc = obj.get('location', {})
    return {"lat": loc.get('latitude'),
            "lng": loc.get('longitude')}

  def get_openNow(self, obj):
    return obj.get('currentOpeningHours', {}).get('openNow')

UserAPIKey — model

  • secrets.token_hex(32) → 64 hex chars
  • Auto-generada en save()
  • Único en BD (unique=True)

hasAPIKey — permission

  • Lee ?ApiKey= de la query
  • Busca en BD el token
  • Inyecta request.user
  • Error descriptivo en español

RegenerateAPIKeyView

POST /api/regen/ → genera un nuevo token_hex(32) y lo guarda. Requiere sesión activa (IsAuthenticated).

api/models.py + api/permissions.py
# ── api/models.py ─────────────────────────────────────
import secrets
from django.db import models
from django.contrib.auth.models import User

class UserAPIKey(models.Model):
  user = models.OneToOneField(
      User,
      on_delete=models.CASCADE,
      related_name="api_key"
  )
  key = models.CharField(max_length=64, unique=True, blank=True)

  def save(self, *args, **kwargs):
    if not self.key:
        # 32 bytes → 64 caracteres hexadecimales
        # secrets es criptográficamente seguro
        self.key = secrets.token_hex(32)
    super().save(*args, **kwargs)


# ── api/permissions.py ────────────────────────────────
from rest_framework.permissions import BasePermission
from api.models import UserAPIKey

class hasAPIKey(BasePermission):
  message = ("Para acceder a este recurso, debes usar "
             "un api_key válido asociado a tu cuenta.")

  def has_permission(self, request, view):
    token = request.query_params.get('ApiKey', '')

    if not token:
        return False       # sin token → 403

    try:
        api_key      = UserAPIKey.objects.get(key=token)
        request.user = api_key.user  # inyectar usuario
        return True
    except UserAPIKey.DoesNotExist:
        return False       # token inválido → 403


# ── api/views.py — RegenerateAPIKeyView ──────────────
class RegenerateAPIKeyView(APIView):
  permission_classes = [permissions.IsAuthenticated]

  def post(self, request):
    api_key     = request.user.api_key
    api_key.key = secrets.token_hex(32)
    api_key.save()
    return Response({"api_key": api_key.key})

¿Por qué Serializers y Django REST?

Decisiones técnicas razonadas

Serializers

  • Validación automática de entrada (registro/login)
  • Transformación de datos anidados (Places)
  • Control de campos expuestos (sin password)
  • Reutilización en varias vistas

Django REST Framework

  • Response() → JSON automático
  • permission_classes → permisos composables
  • CreateAPIView, RetrieveAPIView → CRUD gratis
  • Integración con auth de Django
  • Extensible: hasAPIKey es nuestro permiso propio

Comparativa

Sin DRFCon DRF
json.dumps() manualResponse() automático
if/else de permisospermission_classes
Validar a manoserializer.is_valid()
CRUD desde ceroCreateAPIView...
